//mahasiswa.dart di lib/models

class Mahasiswa {
  String nim;
  String nama;
  String alamat;
  String email;
  String noHp;
  double ipk;
  String foto; // path gambar profil

  Mahasiswa({
    required this.nim,
    required this.nama,
    required this.alamat,
    required this.email,
    required this.noHp,
    required this.ipk,
    required this.foto,
  });
}

//nilai_mahasiswa.dart di lib/models

import 'mahasiswa.dart';

class NilaiMahasiswa {
  Mahasiswa mahasiswa;
  double nilaiUts;
  double nilaiTugas;
  double nilaiUas;
  double pNilaiUts = 0;
  double pNilaiTugas = 0;
  double pNilaiUas = 0;
  double nilaiAkhir = 0;
  String nHuruf = '';
  String predikat = '';

  NilaiMahasiswa({
    required this.mahasiswa,
    required this.nilaiUts,
    required this.nilaiTugas,
    required this.nilaiUas,
  });

  void hitungNilai() {
    pNilaiUts = 0.3 * nilaiUts;
    pNilaiTugas = 0.35 * nilaiTugas;
    pNilaiUas = 0.35 * nilaiUas;
    nilaiAkhir = pNilaiUts + pNilaiTugas + pNilaiUas;
    nHuruf = getNilaiHuruf();
    predikat = getPredikat();
  }

  String getNilaiHuruf() {
    if (nilaiAkhir >= 85) return 'A';
    else if (nilaiAkhir >= 75) return 'B';
    else if (nilaiAkhir >= 65) return 'C';
    else if (nilaiAkhir >= 50) return 'D';
    else return 'E';
  }

  String getPredikat() {
    switch (nHuruf) {
      case 'A':
        return 'Sangat Baik';
      case 'B':
        return 'Baik';
      case 'C':
        return 'Cukup';
      case 'D':
        return 'Kurang';
      default:
        return 'Gagal';
    }
  }
}

//main.dart

import 'package:flutter/material.dart';
import 'models/mahasiswa.dart';
import 'models/nilai_mahasiswa.dart';

void main() {
  runApp(NilaiApp());
}

class NilaiApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Nilai Mahasiswa',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(primarySwatch: Colors.teal),
      home: HitungNilaiPage(),
    );
  }
}

class HitungNilaiPage extends StatefulWidget {
  @override
  State<HitungNilaiPage> createState() => _HitungNilaiPageState();
}

class _HitungNilaiPageState extends State<HitungNilaiPage> {
  final _formKey = GlobalKey<FormState>();
  final nimController = TextEditingController();
  final namaController = TextEditingController();
  final alamatController = TextEditingController();
  final emailController = TextEditingController();
  final noHpController = TextEditingController();
  final ipkController = TextEditingController();
  final utsController = TextEditingController();
  final tugasController = TextEditingController();
  final uasController = TextEditingController();

  final List<NilaiMahasiswa> daftarNilai = [];

  void hitungDanSimpan() {
    if (_formKey.currentState!.validate()) {
      var mhs = Mahasiswa(
        nim: nimController.text,
        nama: namaController.text,
        alamat: alamatController.text,
        email: emailController.text,
        noHp: noHpController.text,
        ipk: double.tryParse(ipkController.text) ?? 0,
        foto: 'assets/profile.jpg',
      );

      var nilai = NilaiMahasiswa(
        mahasiswa: mhs,
        nilaiUts: double.parse(utsController.text),
        nilaiTugas: double.parse(tugasController.text),
        nilaiUas: double.parse(uasController.text),
      );

      nilai.hitungNilai();

      setState(() {
        daftarNilai.add(nilai);
      });

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Data mahasiswa berhasil disimpan!')),
      );

      _formKey.currentState!.reset();
    }
  }

  void lihatDetail(NilaiMahasiswa nilai) {
    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        title: Text('Detail Nilai ${nilai.mahasiswa.nama}'),
        content: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text('NIM: ${nilai.mahasiswa.nim}'),
            Text('Nilai UTS: ${nilai.nilaiUts}'),
            Text('Nilai Tugas: ${nilai.nilaiTugas}'),
            Text('Nilai UAS: ${nilai.nilaiUas}'),
            const Divider(),
            Text('Nilai Akhir: ${nilai.nilaiAkhir.toStringAsFixed(2)}'),
            Text('Nilai Huruf: ${nilai.nHuruf}'),
            Text('Predikat: ${nilai.predikat}'),
          ],
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context),
            child: const Text('Tutup'),
          ),
        ],
      ),
    );
  }

  Widget buildInputField(TextEditingController controller, String label,
      {TextInputType type = TextInputType.text}) {
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 5),
      child: TextFormField(
        controller: controller,
        keyboardType: type,
        decoration: InputDecoration(
          labelText: label,
          border: const OutlineInputBorder(),
        ),
        validator: (v) => v!.isEmpty ? 'Masukkan $label' : null,
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Hitung Nilai Mahasiswa')),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(16),
        child: Column(
          children: [
            Form(
              key: _formKey,
              child: Column(
                children: [
                  const Text(
                    'Input Data Mahasiswa',
                    style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
                  ),
                  buildInputField(nimController, 'NIM'),
                  buildInputField(namaController, 'Nama'),
                  buildInputField(alamatController, 'Alamat'),
                  buildInputField(emailController, 'Email'),
                  buildInputField(noHpController, 'No. HP'),
                  buildInputField(ipkController, 'IPK',
                      type: TextInputType.number),
                  const Divider(),
                  const Text(
                    'Input Nilai',
                    style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
                  ),
                  buildInputField(utsController, 'Nilai UTS',
                      type: TextInputType.number),
                  buildInputField(tugasController, 'Nilai Tugas',
                      type: TextInputType.number),
                  buildInputField(uasController, 'Nilai UAS',
                      type: TextInputType.number),
                  const SizedBox(height: 10),
                  ElevatedButton.icon(
                    icon: const Icon(Icons.save),
                    label: const Text('Hitung & Simpan'),
                    onPressed: hitungDanSimpan,
                  ),
                ],
              ),
            ),
            const SizedBox(height: 20),
            const Divider(thickness: 1),
            const Text(
              'Daftar Nilai Mahasiswa',
              style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 10),
            if (daftarNilai.isEmpty) const Text('Belum ada data mahasiswa'),
            if (daftarNilai.isNotEmpty)
              ListView.builder(
                shrinkWrap: true,
                physics: const NeverScrollableScrollPhysics(),
                itemCount: daftarNilai.length,
                itemBuilder: (context, index) {
                  final data = daftarNilai[index];
                  return Card(
                    color: Colors.teal.shade50,
                    child: ListTile(
                      leading: CircleAvatar(
                        backgroundImage: AssetImage(data.mahasiswa.foto),
                      ),
                      title: Text(data.mahasiswa.nama),
                      subtitle: Text('NIM: ${data.mahasiswa.nim}'),
                      trailing: Text(
                        data.nilaiAkhir.toStringAsFixed(1),
                        style: const TextStyle(fontWeight: FontWeight.bold),
                      ),
                      onTap: () => lihatDetail(data),
                    ),
                  );
                },
              ),
          ],
        ),
      ),
    );
  }
}
